Bienvenido al sistema de pasaportes de Como.

Para comenzar, necesitas configurar la conexión con Google Apps Script:

  1. Crea una nueva hoja de cálculo en Google Sheets
  2. Ve a Extensiones → Apps Script
  3. Copia y pega el siguiente código en el editor:
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  if (data.action === "create") {
    sheet.appendRow([
      new Date(),
      data.dni,
      data.nombre,
      data.apellidos,
      data.fechaNacimiento,
      data.lugarNacimiento,
      data.nacionalidad,
      data.sexo,
      data.fechaExpedicion,
      data.fechaVencimiento,
      data.numeroPasaporte
    ]);
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  if (data.action === "read") {
    var rows = sheet.getDataRange().getValues();
    var headers = rows[0];
    var result = [];
    for (var i = 1; i < rows.length; i++) {
      var obj = {};
      for (var j = 0; j < headers.length; j++) {
        obj[headers[j]] = rows[i][j];
      }
      result.push(obj);
    }
    return ContentService.createTextOutput(JSON.stringify({status: "success", data: result}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
